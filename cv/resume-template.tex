% Ink-skipping underlines for hyperlinks. White artifact copies mask the rule
% around descenders while the blue glyphs remain searchable and selectable.
\newcommand{\artifactwhite}[1]{%
  \special{pdf:code /Artifact BMC}%
  \textcolor{white}{#1}%
  \special{pdf:code EMC}%
}

\makeatletter
\newcommand{\linkunderline}[1]{%
  \begingroup
  \setulcolor{linkblue}%
  \setul{0.18ex}{0.45pt}%
  \setuloverlap{0.20pt}%
  \def\SOUL@uleverysyllable{%
    \setbox0=\hbox{\the\SOUL@syllable}%
    \ifdim\dp0>\z@
      \SOUL@ulunderline{\phantom{\the\SOUL@syllable}}%
      \llap{\artifactwhite{\the\SOUL@syllable}}%
      \llap{\artifactwhite{\the\SOUL@syllable}\hspace{0.18pt}}%
      \llap{\artifactwhite{\the\SOUL@syllable}\hspace{-0.18pt}}%
      \llap{%
        \the\SOUL@syllable
        \SOUL@setkern\SOUL@charkern
      }%
    \else
      \SOUL@ulunderline{%
        \the\SOUL@syllable
        \SOUL@setkern\SOUL@charkern
      }%
    \fi
  }%
  \textul{#1}%
  \endgroup
}
